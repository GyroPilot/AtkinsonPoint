type
   SlideArr = ARRAY[0..picBytes - 1] of -128..127;
   { THE BUS-ERROR FIX (found by register probe, Aug 24 2026).  new()   }
   { can land this buffer at the very BASE of a fresh MMU data segment. }
   { QuickDraw's blitter reads longwords spanning the first and last    }
   { pixels of a non-aligned transfer, reaching a couple of bytes       }
   { BELOW baseAddr - below the segment floor, where nothing is mapped: }
   { Level 7, data seg, offset 65534 (= -2), on the first CopyBits.     }
   { The pads give the blitter mapped slack on both sides, wherever     }
   { the allocator puts the block.                                       }
   SlideBuf = record
                 padLo : longint;
                 a     : SlideArr;
                 padHi : longint
              end;
   SlidePtr = ^SlideBuf;

   { QuickDraw owns Picture / PicPtr / PicHandle, hence the names above. }
   { Lisa Pascal has no type casts, so a variant record is how one       }
   { address is handed to QuickDraw as a QDPtr and to READ_DATA as a     }
   { LONGINT.                                                            }
   PtrCast = record
               case integer of
                 1: (pp : SlidePtr);
                 2: (qp : QDPtr);
                 3: (ln : longint)
             end;

IMPLEMENTATION

procedure HandleFilerOp {var wantsOpen : boolean};
var fx  : FilerExt;
    err : integer;
begin
   wantsOpen := false;
   GetAddParms(err, theEvent, fx);

   case fx.theFlrOp of

      fcNone :
         wantsOpen := true;              { the user opened the tool }

      fcResume :
         TellFiler(err, docClosd, internalError, theEvent.who);

      fcClose, fcSuspend, fcShred :
         begin
            myWindow := nil;
            haveShown := false;
            TellFiler(err, docClosd, allOK, theEvent.who)
         end;

      fcCopy, fcPut :
         TellFiler(err, docNotXfered, internalError, theEvent.who);

      fcDfClose :
         TellFiler(err, dfClosed, allOK, theEvent.who);

      fcTerminate :
         toldToDie := true;

      otherwise

   end
end;
